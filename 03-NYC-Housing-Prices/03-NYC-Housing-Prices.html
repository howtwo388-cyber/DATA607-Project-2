Approach

Missing Values Analysis

Code 
# Check missing values in the five borough columns
missing_summary <- housing %>%
  select(Bronx, Brooklyn, Manhattan, Queens, `Staten Island`) %>%
  summarise(
    across(
      everything(),
      list(
        Missing = ~sum(is.na(.x)),
        Available = ~sum(!is.na(.x))
      )
    )
  ) %>%
  pivot_longer(
    everything(),
    names_to = c("Borough", ".value"),
    names_pattern = "^(.*)_(Missing|Available)$"
  ) %>%
  mutate(
    Missing_Percent = round(
      100 * Missing / (Missing + Available), 1
    )
  )

Data Cleaning and Transformation

The original dataset stores the five boroughs in separate columns.

To make the data easier to analyze, the borough columns are transformed from wide to long format using pivot_longer().

Code 
# Create a clean dataset
housing_clean <- housing %>%
  select(
    Year, Month,
    Bronx, Brooklyn, Manhattan,
    Queens, `Staten Island`
  ) %>%
  mutate(
    Year = as.integer(Year),
    Month = factor(
      Month,
      levels = month.abb,
      ordered = TRUE
    )
  )

# Transform wide data into long format
housing_long <- housing_clean %>%
  pivot_longer(
    cols = c(
      Bronx, Brooklyn, Manhattan,
      Queens, `Staten Island`
    ),
    names_to = "Borough",
    values_to = "Average_Price"
  ) %>%
  filter(!is.na(Average_Price)) %>%
  mutate(
    Date = as.Date(
      paste(Year, as.integer(Month), "01", sep = "-")
    )
  )

Question 2: Which New York City borough has experienced the greatest change in average housing prices over time?

The second business question examines how housing prices changed between the first and last available months.

The percentage change is calculated using the first and last recorded prices for each borough.

Code 
# Calculate the change in housing prices
price_change <- housing_long %>%
  arrange(Borough, Date) %>%
  group_by(Borough) %>%
  summarise(
    First_Date = first(Date),
    Last_Date = last(Date),
    First_Price = first(Average_Price),
    Last_Price = last(Average_Price),
    Price_Change = Last_Price - First_Price,
    Percent_Change = if (
      first(Average_Price) != 0
    ) {
      100 * (
        last(Average_Price) - first(Average_Price)
      ) / first(Average_Price)
    } else {
      NA_real_
    },
    .groups = "drop"
  ) %>%
  arrange(desc(Percent_Change))
